""" RetailGraph -> Saree Sale Tracker sync script ----------------------------------------------- Reads NEW sales bills from RetailGraph's SQL Server (read-only) and creates matching "unclaimed" entries in the Saree Sale Tracker's Firestore database. Salesman is left blank on purpose -- staff claim their own bills inside the app with one tap. This script only ever runs SELECT queries against SQL Server. It never writes, updates, or deletes anything in RetailGraph. Uses plain HTTPS (via the "requests" library) to talk to Firestore, instead of Google's grpc library -- this avoids connection issues some networks/antivirus/firewalls have with grpc. SETUP (one-time): 1. pip install pyodbc requests google-auth 2. Download your Firebase service account key: Firebase console -> gear icon -> Project settings -> Service accounts -> Generate new private key -> save as serviceAccountKey.json in this same folder. 3. Fill in the SQL_* settings below with your read-only login. 4. Run once manually to test: python sync_retailgraph.py 5. Once confirmed working, schedule it in Windows Task Scheduler to run every 15-30 minutes. """ import pyodbc import requests from google.oauth2 import service_account import google.auth.transport.requests from datetime import datetime, timezone import os # ---------------- SETTINGS ---------------- SQL_SERVER = r"DESKTOP-IJS0LSO\SQLEXPRESS" SQL_DATABASE = "BHABHINEW" SQL_USERNAME = "saree_tracker_readonly" # the read-only login created earlier SQL_PASSWORD = "Bhabhi@2026" # <-- your actual password (already set) # Only bills from the last N days are checked on each run -- keeps the # query fast. Increase this if you ever need to backfill further. SYNC_WINDOW_DAYS = 3 EMAIL_DOMAIN = "@bhabhisareestracker.app" ADMIN_EMAIL = "mohit" + EMAIL_DOMAIN # Known staff -- ReferBy names are matched against this list (case/spacing # insensitive). Anything that doesn't match stays Unclaimed rather than # risk assigning a sale to the wrong person. KNOWN_STAFF = ["Jitu", "Vicky", "Hitesh", "Chotu", "Manish", "Purab"] def match_salesman(referby_name): """Returns (properName, email) if the ReferBy name matches a known staff member or the owner, otherwise (None, None).""" if not referby_name: return (None, None) cleaned = referby_name.strip().lower() if cleaned == "mohit": return ("Mohit", ADMIN_EMAIL) for name in KNOWN_STAFF: if name.lower() == cleaned: return (name, name.lower() + EMAIL_DOMAIN) return (None, None) SCRIPT_DIR = os.path.dirname(os.path.abspath(__file__)) SERVICE_ACCOUNT_PATH = os.path.join(SCRIPT_DIR, "serviceAccountKey.json") # ------------------------------------------- def fetch_recent_bills(): conn_str = ( f"DRIVER={{ODBC Driver 17 for SQL Server}};" f"SERVER={SQL_SERVER};DATABASE={SQL_DATABASE};" f"UID={SQL_USERNAME};PWD={SQL_PASSWORD}" ) conn = pyodbc.connect(conn_str) cursor = conn.cursor() # READ-ONLY: this is a SELECT query only. Nothing is written to RetailGraph. # Always checks the last 7 days fresh on every run -- this is what lets us # catch invoices that were deleted and re-entered, not just brand-new ones. query = """ SELECT t.PKID AS SaleID, t.EntryNo AS InvoiceNumber, t.EntryDate AS SaleDate, c.Customer AS CustomerName, c.Phone1 AS CustomerPhone, p.Product AS SareeName, t.NetAmt AS TotalAmount, t.Credit AS IsCredit, t.CreditAmt AS CreditAmount, t.CreditDate AS CreditDueDate, rm.ReferBy AS SalesmanName FROM tblSalesInv_Trn t INNER JOIN tblSalesInv_Dtl d ON d.FKID = t.PKID LEFT JOIN tblCustomer_Mas c ON c.PKCustomerID = t.FKCustomerID LEFT JOIN tblProd_Mas p ON p.PKProdId = d.FKProdID LEFT JOIN tblReferBy_Lnk rl ON rl.FKID = t.PKID LEFT JOIN tblReferBy_Mas rm ON rm.PKReferId = rl.FKReferByID WHERE t.EntryDate >= DATEADD(day, -?, CAST(GETDATE() AS DATE)) ORDER BY t.PKID ASC """ cursor.execute(query, SYNC_WINDOW_DAYS) rows = cursor.fetchall() conn.close() return rows def get_access_token(): creds = service_account.Credentials.from_service_account_file( SERVICE_ACCOUNT_PATH, scopes=["https://www.googleapis.com/auth/datastore"] ) creds.refresh(google.auth.transport.requests.Request()) return creds.token, creds.project_id def fv(value): """Convert a Python value into a Firestore REST 'value' object.""" if value is None: return {"nullValue": None} if isinstance(value, bool): return {"booleanValue": value} if isinstance(value, int): return {"integerValue": str(value)} if isinstance(value, float): return {"doubleValue": value} return {"stringValue": str(value)} def push_to_firestore(token, project_id, rows): if not rows: print("No bills found in the last 7 days.") return # Group line items by invoice (SaleID) -- one bill can have several sarees invoices = {} order = [] for row in rows: sale_id, invoice_no, sale_date, customer, phone, saree, total_amt, is_credit, credit_amt, credit_due, referby_name = row if sale_id not in invoices: salesman_name, salesman_email = match_salesman(referby_name) invoices[sale_id] = { "sale_id": sale_id, "invoice_no": invoice_no, "sale_date": sale_date, "customer": customer, "phone": phone, "items": [], "total_amt": float(total_amt) if total_amt is not None else None, "credit_amt": float(credit_amt) if (is_credit and credit_amt) else None, "credit_due": credit_due, "salesman_name": salesman_name, "salesman_email": salesman_email } order.append(sale_id) invoices[sale_id]["items"].append(saree or "Unknown item") base_url = f"https://firestore.googleapis.com/v1/projects/{project_id}/databases/(default)/documents/sareeSales" headers = {"Authorization": f"Bearer {token}", "Content-Type": "application/json"} created = 0 refreshed = 0 assigned = 0 unmatched_names = set() # These are the only fields RetailGraph is allowed to overwrite on an # already-synced invoice. Claim status, packing status, and photos # belong to the app/staff from here on and are never touched again. billing_fields = ["invoice", "items", "customer", "phone", "saleDate", "totalAmount", "creditAmount", "creditDue"] for sale_id in order: inv = invoices[sale_id] doc_id = f"rg_{sale_id}" doc_url = f"{base_url}/{doc_id}" billing_data = { "invoice": fv(str(inv["invoice_no"])), "items": {"arrayValue": {"values": [fv(i) for i in inv["items"]]}}, "customer": fv(inv["customer"]), "phone": fv(inv["phone"]), "saleDate": fv(inv["sale_date"].strftime("%Y-%m-%d")), "totalAmount": fv(inv["total_amt"]), "creditAmount": fv(inv["credit_amt"]), "creditDue": fv(inv["credit_due"].strftime("%Y-%m-%d") if inv["credit_due"] else None), } # Check if this invoice already exists in the tracker check = requests.get(doc_url, headers=headers) if check.status_code == 200: # Already synced before -- refresh only billing details, # leave salesman/status/photo exactly as staff left them. mask = "&".join(f"updateMask.fieldPaths={f}" for f in billing_fields) resp = requests.patch(f"{doc_url}?{mask}", headers=headers, json={"fields": billing_data}) if resp.status_code < 300: refreshed += 1 else: print(f"Failed to refresh invoice {inv['invoice_no']}: {resp.status_code} {resp.text}") else: # Brand new invoice -- create it as Unclaimed full_data = dict(billing_data) full_data.update({ "salesman": fv(inv["salesman_name"]), "salesmanEmail": fv(inv["salesman_email"]), "status": fv("Sold"), "photoUrl": fv(None), "deliveryMethod": fv(None), "source": fv("retailgraph_auto"), "retailGraphSaleID": fv(sale_id), }) resp = requests.patch(doc_url, headers=headers, json={"fields": full_data}) if resp.status_code < 300: created += 1 if inv["salesman_name"]: assigned += 1 else: print(f"Failed to create invoice {inv['invoice_no']}: {resp.status_code} {resp.text}") print(f"{created} new invoice(s) added ({assigned} auto-assigned to a salesman), {refreshed} existing invoice(s) refreshed.") def main(): if not os.path.exists(SERVICE_ACCOUNT_PATH): print(f"ERROR: {SERVICE_ACCOUNT_PATH} not found. See setup instructions at the top of this file.") return token, project_id = get_access_token() print(f"[{datetime.now()}] Checking RetailGraph for bills from the last {SYNC_WINDOW_DAYS} days...") rows = fetch_recent_bills() push_to_firestore(token, project_id, rows) if __name__ == "__main__": main()
